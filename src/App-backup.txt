import { useState } from "react";

const projects = [
  {
    name: "Pepsi",
    total: 500,
    interview: 350,
    hired: 140,
    processing: 150,
  },
  {
    name: "Samsung",
    total: 430,
    interview: 300,
    hired: 110,
    processing: 140,
  },
  {
    name: "Electrolux",
    total: 320,
    interview: 200,
    hired: 70,
    processing: 110,
  },
];

function App() {
  const [selectedProject, setSelectedProject] = useState("All");

  const filteredProjects =
    selectedProject === "All"
      ? projects
      : projects.filter((p) => p.name === selectedProject);

  const kpis = [
    { title: "Tổng ứng viên", field: "total", color: "#2563eb" },
    { title: "Đã phỏng vấn", field: "interview", color: "#7c3aed" },
    { title: "Nhận việc", field: "hired", color: "#16a34a" },
    { title: "Đang xử lý", field: "processing", color: "#d97706" },
  ] as const;
const maxTotal = Math.max(
  1,
  ...projects.map((p) => p.total)
);
  return (
    <div style={{ padding: 32, fontFamily: "Arial" }}>
      <h1>Staffing Dashboard</h1>
      <p>Báo cáo kết quả tuyển dụng</p>

      <label htmlFor="project">Chọn dự án: </label>
      <select
        id="project"
        value={selectedProject}
        onChange={(e) => setSelectedProject(e.target.value)}
        style={{ padding: 10, margin: 12, borderRadius: 6 }}
      >
        <option value="All">Tất cả dự án</option>
        {projects.map((p) => (
          <option key={p.name} value={p.name}>
            {p.name}
          </option>
        ))}
      </select>

      <div style={{
        display: "grid",
        gridTemplateColumns: "repeat(auto-fit, minmax(200px, 1fr))",
        gap: 16,
        marginTop: 24,
      }}>
        {kpis.map((kpi) => {
          const value = filteredProjects.reduce(
            (sum, p) => sum + p[kpi.field], 0
          );

          return (
            <div key={kpi.title} style={{
              padding: 24,
              background: "#fff",
              border: "1px solid #e5e7eb",
              borderRadius: 12,
              boxShadow: "0 2px 6px rgba(0,0,0,0.05)",
            }}>
              <p style={{ color: "#64748b", margin: 0 }}>
                {kpi.title}
              </p>
              <h2 style={{ fontSize: 30, margin: "12px 0" }}>
                {value.toLocaleString("vi-VN")}
              </h2>
              <div style={{
                background: kpi.color,
                height: 4,
                borderRadius: 4,
              }} />
            </div>
          );
        })}
      </div>{/* BIỂU ĐỒ TUYỂN DỤNG */}
<div
  style={{
    marginTop: 32,
    padding: 24,
    background: "#ffffff",
    border: "1px solid #e5e7eb",
    borderRadius: 12,
  }}
>
  <h2>Kết quả tuyển dụng theo dự án</h2>

  {filteredProjects.map((p) => (
    <div key={p.name} style={{ marginBottom: 28 }}>
      <h3>{p.name}</h3>

      <p>Tổng ứng viên: {p.total}</p>
      <div style={{ background: "#f1f5f9", borderRadius: 6 }}>
        <div
          style={{
            width: `${(p.total / maxTotal) * 100}%`,
            height: 22,
            background: "#2563eb",
            borderRadius: 6,
          }}
        />
      </div>

      <p>Nhận việc: {p.hired}</p>
      <div style={{ background: "#f1f5f9", borderRadius: 6 }}>
        <div
          style={{
            width: `${(p.hired / maxTotal) * 100}%`,
            height: 22,
            background: "#16a34a",
            borderRadius: 6,
          }}
        />
      </div>
    </div>
  ))}
</div>
    </div>
  );
}

export default App;
